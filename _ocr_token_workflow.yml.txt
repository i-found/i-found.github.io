name: 刷新百度 OCR token

# 每天检查一次 ocr_token.json 的剩余有效期，不足 15 天才真正换新，
# 换完自动提交回仓库，页面那边什么都不用做。
on:
  schedule:
    - cron: '0 4 * * *' # 每天 04:00 UTC（北京时间 12:00）
  workflow_dispatch: # 也可以在 Actions 页面手动点一次

permissions:
  contents: write

concurrency:
  group: refresh-ocr-token
  cancel-in-progress: true

jobs:
  refresh:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: 检查剩余有效期
        id: check
        run: |
          NEED=$(python3 -c "import json,os,time;d=json.load(open('ocr_token.json',encoding='utf-8')) if os.path.isfile('ocr_token.json') else {};print('yes' if int(d.get('expires_at',0))-time.time() < 15*86400 else 'no')")
          echo "need=$NEED" >> "$GITHUB_OUTPUT"
          echo "需要刷新: $NEED"

      - name: 换取新 token
        if: steps.check.outputs.need == 'yes'
        env:
          BAIDU_AK: ${{ secrets.BAIDU_AK }}
          BAIDU_SK: ${{ secrets.BAIDU_SK }}
        run: |
          if [ -z "$BAIDU_AK" ] || [ -z "$BAIDU_SK" ]; then
            echo "没有配置 BAIDU_AK / BAIDU_SK，跳过。请到 Settings → Secrets and variables → Actions 添加。"
            exit 0
          fi
          python3 _refresh_ocr_token.py

      - name: 提交新 token
        if: steps.check.outputs.need == 'yes'
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add ocr_token.json
          if git diff --cached --quiet; then
            echo "token 没有变化，不用提交"
            exit 0
          fi
          git commit -m "chore: 自动刷新百度 OCR token"
          git push || (git pull --rebase --autostash && git push)
